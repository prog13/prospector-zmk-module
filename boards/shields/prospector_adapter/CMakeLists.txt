  file(GLOB font_sources src/fonts/*.c)
  zephyr_library_sources(${ZEPHYR_BASE}/misc/empty_file.c)
  zephyr_library_include_directories(${ZEPHYR_LVGL_MODULE_DIR})
  zephyr_library_include_directories(${ZEPHYR_BASE}/lib/gui/lvgl/)
  zephyr_library_include_directories(${ZEPHYR_BASE}/drivers)
  zephyr_library_include_directories(${CMAKE_SOURCE_DIR}/include)
  zephyr_library_include_directories(${ZEPHYR_CURRENT_MODULE_DIR}/include)
  zephyr_library_include_directories(${ZEPHYR_CURRENT_CMAKE_DIR}/include)
  zephyr_library_include_directories(include)
  zephyr_library_sources(src/brightness.c)
  zephyr_library_sources(src/custom_status_screen.c)
  zephyr_library_sources(src/display_rotate_init.c)
  zephyr_library_sources_ifdef(CONFIG_CDC_ACM_DTE_RATE_CALLBACK_SUPPORT src/usb_dfu_touch.c)

  if(CONFIG_PROSPECTOR_TOUCH_BRIGHTNESS)
    zephyr_library_sources(src/touch_brightness_drag.c)
    # Ferro gates the drag itself, in layouts/ferro/touch_pole.c.
    if(NOT CONFIG_PROSPECTOR_STATUS_SCREEN_FERRO)
      zephyr_library_sources(src/touch_brightness.c)
    endif()
  endif()

  if(CONFIG_PROSPECTOR_TOUCH_DEBUG)
    zephyr_library_sources(src/touch_debug.c)
  endif()

  if(CONFIG_PROSPECTOR_SHOW_MODIFIERS)
    zephyr_library_sources(src/modifier_order.c)
  endif()

  if(CONFIG_PROSPECTOR_STATUS_SCREEN_CLASSIC)
    file(GLOB layout_sources src/layouts/classic/*.c)
    list(FILTER layout_sources EXCLUDE REGEX "status_screen\\.c$")
    zephyr_library_sources(${layout_sources})
  elseif(CONFIG_PROSPECTOR_STATUS_SCREEN_RADII)
    file(GLOB layout_sources src/layouts/radii/*.c)
    list(FILTER layout_sources EXCLUDE REGEX "status_screen\\.c$")
    zephyr_library_sources(${layout_sources})
  elseif(CONFIG_PROSPECTOR_STATUS_SCREEN_FIELD)
    file(GLOB layout_sources src/layouts/field/*.c)
    zephyr_library_sources(src/wpm_idle.c)
    list(FILTER layout_sources EXCLUDE REGEX "status_screen\\.c$")
    zephyr_library_sources(${layout_sources})
  elseif(CONFIG_PROSPECTOR_STATUS_SCREEN_FLUX)
    file(GLOB layout_sources src/layouts/flux/*.c)
    zephyr_library_sources(src/wpm_idle.c)
    list(FILTER layout_sources EXCLUDE REGEX "(status_screen|touch_pole)\\.c$")
    zephyr_library_sources(${layout_sources})
    if(CONFIG_PROSPECTOR_TOUCH_FIELD_POLE)
      zephyr_library_sources(src/layouts/flux/touch_pole.c)
    endif()
  elseif(CONFIG_PROSPECTOR_STATUS_SCREEN_FERRO)
    file(GLOB layout_sources src/layouts/ferro/*.c)
    zephyr_library_sources(src/wpm_idle.c)
    list(FILTER layout_sources EXCLUDE REGEX
         "(status_screen|touch_pole|touch_palette|brightness_readout)\\.c$")
    zephyr_library_sources(${layout_sources})
    zephyr_library_sources(src/layouts/flux/flux_poles.c)
    # These two are per-pixel hot loops; at the default -Os they miss frames.
    set_source_files_properties(src/layouts/ferro/ferro_blobs.c src/layouts/flux/flux_poles.c
                                PROPERTIES COMPILE_OPTIONS "-O2")
    if(CONFIG_PROSPECTOR_TOUCH_FIELD_POLE OR CONFIG_PROSPECTOR_TOUCH_BRIGHTNESS)
      zephyr_library_sources(src/layouts/ferro/touch_pole.c)
    endif()
    if(CONFIG_PROSPECTOR_TOUCH_BRIGHTNESS)
      zephyr_library_sources(src/touch_brightness_gate.c)
      zephyr_library_sources(src/layouts/ferro/brightness_readout.c)
    endif()
    if(CONFIG_PROSPECTOR_FERRO_PALETTE_CYCLE)
      zephyr_library_sources(src/layouts/ferro/touch_palette.c)
    endif()
  elseif(CONFIG_PROSPECTOR_STATUS_SCREEN_OPERATOR)
    file(GLOB layout_sources src/layouts/operator/*.c)
    list(FILTER layout_sources EXCLUDE REGEX "status_screen\\.c$")
    zephyr_library_sources(${layout_sources})
  endif()

  zephyr_library_sources(${font_sources})
